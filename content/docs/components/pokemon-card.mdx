## Usage

```tsx
import { PokemonCard, PokemonCardHeader, PokemonCardTitle } from "@/components/compodex/ui/pokemon-card"
```

```tsx
<PokemonCard type="fire">
  <PokemonCardHeader>
    <PokemonCardTitle>Charmander</PokemonCardTitle>
  </PokemonCardHeader>
</PokemonCard>
```
